## 06-03. 생성

수집하고 분석한 정보를 바탕으로 전문적이고 구조화된 최종 보고서를 생성합니다. 독자가 이해하기 쉽고 실용적인 가치를 제공하는 보고서 작성 방법을 학습합니다.

### 학습 목표
- **보고서 구조 설계**: 논리적이고 체계적인 보고서 구조 설계
- **내용 통합**: 수집된 정보를 일관성 있게 통합
- **인사이트 도출**: 단순한 정보 나열을 넘어서 통찰 제공
- **최종 검토**: 보고서의 완성도와 정확성 검증

### 2. 내용 매핑
수집된 정보를 보고서 구조에 매핑합니다:

```python
def map_content_to_structure(categories: list, insights: list, 
                           research_goals: dict) -> dict:
    """내용을 보고서 구조에 매핑"""
    content_mapping = {
        "요약": [],
        "서론": [],
        "배경": [],
        "주요 발견사항": [],
        "상세 분석": [],
        "결론 및 권고사항": []
    }
    
    # 카테고리별 내용 매핑
    for category in categories:
        if "정의" in category.category or "개념" in category.category:
            content_mapping["배경"].extend(category.information)
        elif "현재 상황" in category.category:
            content_mapping["주요 발견사항"].extend(category.information)
        elif "미래" in category.category or "전망" in category.category:
            content_mapping["결론 및 권고사항"].extend(category.information)
        else:
            content_mapping["상세 분석"].extend(category.information)
    
    # 인사이트를 적절한 섹션에 배치
    content_mapping["주요 발견사항"].extend(insights)
    
    return content_mapping
```

## 내용 통합

### 1. 정보 통합 및 정리
수집된 정보를 일관성 있게 통합합니다:

```python
def integrate_information(content_mapping: dict) -> dict:
    """정보 통합 및 정리"""
    integrated_content = {}
    
    for section, content_list in content_mapping.items():
        # 중복 제거 및 정리
        unique_content = list(set(content_list))
        
        # 내용을 논리적 순서로 정렬
        if section == "주요 발견사항":
            # 중요도 순으로 정렬
            sorted_content = sort_by_importance(unique_content)
        elif section == "상세 분석":
            # 주제별로 그룹화
            sorted_content = group_by_topic(unique_content)
        else:
            # 시간순 또는 논리적 순서로 정렬
            sorted_content = sort_logically(unique_content)
        
        integrated_content[section] = sorted_content
    
    return integrated_content

def sort_by_importance(content_list: list) -> list:
    """중요도에 따라 정렬"""
    # 간단한 중요도 점수 계산
    importance_scores = {}
    for content in content_list:
        score = 0
        if "핵심" in content or "중요" in content:
            score += 3
        if "최신" in content or "2024" in content:
            score += 2
        if len(content) > 100:  # 상세한 내용
            score += 1
        importance_scores[content] = score
    
    return sorted(content_list, key=lambda x: importance_scores[x], reverse=True)

def group_by_topic(content_list: list) -> dict:
    """주제별로 그룹화"""
    topics = {}
    
    for content in content_list:
        # 간단한 키워드 기반 분류
        if any(word in content.lower() for word in ["기술", "기능", "특징"]):
            topic = "기술적 측면"
        elif any(word in content.lower() for word in ["시장", "경제", "비즈니스"]):
            topic = "시장 및 비즈니스"
        elif any(word in content.lower() for word in ["사회", "영향", "윤리"]):
            topic = "사회적 영향"
        else:
            topic = "기타"
        
        if topic not in topics:
            topics[topic] = []
        topics[topic].append(content)
    
    return topics
```

### 2. 일관성 검증
통합된 내용의 일관성을 검증합니다:

```python
def verify_content_consistency(integrated_content: dict) -> dict:
    """내용 일관성 검증"""
    consistency_report = {
        "inconsistencies": [],
        "gaps": [],
        "redundancies": [],
        "overall_score": 0.0
    }
    
    # 섹션 간 일관성 검사
    all_content = []
    for section_content in integrated_content.values():
        if isinstance(section_content, list):
            all_content.extend(section_content)
        elif isinstance(section_content, dict):
            for topic_content in section_content.values():
                all_content.extend(topic_content)
    
    # 중복 내용 검사
    content_counts = {}
    for content in all_content:
        content_counts[content] = content_counts.get(content, 0) + 1
    
    for content, count in content_counts.items():
        if count > 1:
            consistency_report["redundancies"].append(content)
    
    # 일관성 점수 계산
    total_content = len(all_content)
    redundant_content = len(consistency_report["redundancies"])
    consistency_score = 1.0 - (redundant_content / total_content) if total_content > 0 else 1.0
    consistency_report["overall_score"] = consistency_score
    
    return consistency_report
```

## 인사이트 도출

### 1. 핵심 인사이트 추출
수집된 정보에서 핵심 인사이트를 도출합니다:

```python
def extract_key_insights(integrated_content: dict, research_goals: dict) -> list:
    """핵심 인사이트 추출"""
    insights = []
    
    # 각 섹션에서 인사이트 추출
    for section, content in integrated_content.items():
        if section == "주요 발견사항":
            # 통계적 인사이트
            statistical_insights = extract_statistical_insights(content)
            insights.extend(statistical_insights)
        
        elif section == "상세 분석":
            # 트렌드 인사이트
            trend_insights = extract_trend_insights(content)
            insights.extend(trend_insights)
        
        elif section == "결론 및 권고사항":
            # 전략적 인사이트
            strategic_insights = extract_strategic_insights(content)
            insights.extend(strategic_insights)
    
    # 인사이트 우선순위 설정
    prioritized_insights = prioritize_insights(insights, research_goals)
    
    return prioritized_insights

def extract_statistical_insights(content: list) -> list:
    """통계적 인사이트 추출"""
    insights = []
    
    for item in content:
        # 숫자나 퍼센트가 포함된 내용에서 인사이트 추출
        if any(char.isdigit() for char in item):
            if "%" in item or "증가" in item or "감소" in item:
                insights.append(f"통계적 발견: {item}")
    
    return insights

def extract_trend_insights(content: dict) -> list:
    """트렌드 인사이트 추출"""
    insights = []
    
    for topic, topic_content in content.items():
        if len(topic_content) > 3:  # 충분한 데이터가 있는 주제
            insights.append(f"{topic}에서 명확한 트렌드 관찰됨")
    
    return insights

def prioritize_insights(insights: list, research_goals: dict) -> list:
    """인사이트 우선순위 설정"""
    # 연구 목표와의 관련성에 따라 우선순위 설정
    prioritized = []
    
    for insight in insights:
        relevance_score = 0
        for goal in research_goals.get("secondary_objectives", []):
            if any(word in insight.lower() for word in goal.lower().split()):
                relevance_score += 1
        
        prioritized.append((insight, relevance_score))
    
    # 관련성 점수에 따라 정렬
    prioritized.sort(key=lambda x: x[1], reverse=True)
    
    return [insight for insight, score in prioritized]
```

### 2. 권고사항 생성
연구 결과를 바탕으로 실행 가능한 권고사항을 생성합니다:

```python
def generate_recommendations(insights: list, research_scope: dict) -> list:
    """권고사항 생성"""
    recommendations = []
    
    # 인사이트 기반 권고사항 생성
    for insight in insights[:5]:  # 상위 5개 인사이트만 사용
        if "증가" in insight or "성장" in insight:
            recommendations.append(f"긍정적 트렌드를 활용한 전략 수립 필요")
        elif "감소" in insight or "위험" in insight:
            recommendations.append(f"리스크 관리 방안 마련 필요")
        elif "기술" in insight:
            recommendations.append(f"기술 투자 및 개발 계획 검토 필요")
        elif "시장" in insight:
            recommendations.append(f"시장 진입 전략 재검토 필요")
    
    # 범위에 따른 추가 권고사항
    if research_scope.get("depth_level") == "심층":
        recommendations.append("추가 연구 및 모니터링 계획 수립")
    
    return recommendations
```

## 최종 보고서 생성

### 1. 보고서 작성
통합된 내용과 인사이트를 바탕으로 최종 보고서를 작성합니다:

```python
def generate_final_report(integrated_content: dict, insights: list, 
                         recommendations: list, research_request: dict) -> str:
    """최종 보고서 생성"""
    report = f"""
# {research_request['topic']} 심층 리서치 보고서

## 요약 (Executive Summary)

본 보고서는 {research_request['topic']}에 대한 심층 분석을 통해 다음과 같은 핵심 발견사항을 제시합니다:

### 주요 발견사항
"""
    
    # 주요 발견사항 추가
    for i, insight in enumerate(insights[:3], 1):
        report += f"{i}. {insight}\n"
    
    report += f"""
### 핵심 권고사항
"""
    
    # 권고사항 추가
    for i, rec in enumerate(recommendations[:3], 1):
        report += f"{i}. {rec}\n"
    
    # 각 섹션별 상세 내용 추가
    for section, content in integrated_content.items():
        if section != "요약":
            report += f"\n## {section}\n\n"
            
            if isinstance(content, list):
                for item in content:
                    report += f"- {item}\n"
            elif isinstance(content, dict):
                for topic, topic_content in content.items():
                    report += f"### {topic}\n"
                    for item in topic_content:
                        report += f"- {item}\n"
    
    report += f"""
## 결론

{research_request['topic']}에 대한 종합적인 분석을 통해 현재 상황과 미래 전망을 제시했습니다. 
제시된 권고사항을 바탕으로 전략적 의사결정에 활용하시기 바랍니다.

---
*본 보고서는 {len(integrated_content)}개 소스에서 수집된 정보를 바탕으로 작성되었습니다.*
"""
    
    return report
```

### 2. 보고서 검토 및 개선
생성된 보고서를 검토하고 개선합니다:

```python
def review_and_improve_report(report: str) -> dict:
    """보고서 검토 및 개선"""
    review_result = {
        "word_count": len(report.split()),
        "section_count": report.count("##"),
        "issues": [],
        "improvements": []
    }
    
    # 기본 검토 항목
    if review_result["word_count"] < 1000:
        review_result["issues"].append("보고서 길이가 너무 짧음")
        review_result["improvements"].append("더 상세한 분석 내용 추가 필요")
    
    if review_result["section_count"] < 4:
        review_result["issues"].append("섹션 구성이 부족함")
        review_result["improvements"].append("추가 섹션 구성 필요")
    
    # 내용 품질 검토
    if "데이터" not in report and "통계" not in report:
        review_result["issues"].append("정량적 데이터 부족")
        review_result["improvements"].append("통계 및 데이터 추가 필요")
    
    return review_result

def finalize_report(report: str, review_result: dict) -> str:
    """보고서 최종 완성"""
    # 개선사항 반영
    if "정량적 데이터 부족" in review_result["issues"]:
        report += "\n\n*참고: 추가 정량적 데이터 수집을 통한 분석 강화 권장*"
    
    # 최종 형식 정리
    final_report = report.strip()
    
    return final_report
```
